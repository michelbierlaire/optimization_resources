\item The derivative of a function of one variable is defined as
  \[
\frac{df(x)}{dx}=f'(x) = \lim_{h \to 0}\frac{f(x+h)-f(x)}{h}
\]
if the limit exists. Calculate the derivative of the following
function using the above definition:
\[
f(x)= \frac{x}{x-1}.
\]
\item Calculate the gradient of the following function:

 \[f(x_1,x_2,x_3)= e^{x_1} + x_1^2x_3 -x_1x_2x_3.\]

\item Consider the function $f$, and calculate
the directional derivative along the direction $d$ at a point $(x_1,
x_2, x_3)$, where
\begin{align}
f(x_1,x_2,x_3)&=  e^{x_1} + x_1^2x_3 -x_1x_2x_3,\\
  d
&=
\begin{pmatrix}
    d_1 \\
    d_2 \\
    d_3
\end{pmatrix}.
\end{align}

\item
 Consider the function $$f(x,y)=x^2 +y^2.$$ The directional derivative
 of f in the direction of the vector $(1,1)$ at the point $(0,0)$ is
 $2\sqrt{2}$. Is it correct? 

\item Calculate the steepest descent direction of the
  following function
  \[
  f(x)=  \frac{1}{2}x_1^2+2x_2^2,
  \]
  at point $x= (1,1)$.

\item  Calculate the Jacobian matrix given the following function:
\[
f(x_1,x_2)= (x_1^2x_2, cos(e^{x_1+x_2^2}).
\]


\item The surface of a lake is given by a region $D$ and a
plane $XY$. It's depth (in meters) in point $(x,y)$ is given by the function: $$f(x,y)=400-3x^2y^2.$$ If a swimmer is located in point $(1,-2)$,determine which direction he needs to swim in order to make the depth increase as quick as possible.

\end{enumerate}
